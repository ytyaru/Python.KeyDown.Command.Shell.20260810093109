#!/usr/bin/env python3
import sys
import glob
from evdev import InputDevice, ecodes

def find_keyboard_device():
    """システム上の有効な物理キーボードデバイスを自動探索"""
    for path in glob.glob('/dev/input/event*'):
        try:
            dev = InputDevice(path)
            if ecodes.KEY_A in dev.capabilities().get(ecodes.EV_KEY, []):
                return dev
        except Exception:
            continue
    return None

def main():
    dev = find_keyboard_device()
    if not dev:
        print("エラー: 物理キーボードデバイスが見つかりません。パスの権限を確認してください。", file=sys.stderr)
        sys.exit(1)

    print(f"--- evdev 定数名調査モード ---", file=sys.stderr)
    print(f"使用デバイス: {dev.name} ({dev.path})", file=sys.stderr)
    print("調べたいキーを押してください。TSV用の定数名が表示されます。（終了: Ctrl+C）\n", file=sys.stderr)

    try:
        # キーボードの排他制御を行い、端末への不要な入力を防ぐ
        dev.grab()
        
        for event in dev.read_loop():
            # キーイベント(EV_KEY) かつ キーが押された瞬間(value=1)
            if event.type == ecodes.EV_KEY and event.value == 1:
                # 物理コードから定数名（文字列）を逆引き
                keycode_name = ecodes.KEY.get(event.code, "UNKNOWN")
                
                # TSVにそのまま使える形式で出力
                print(f"検知された定数名: {keycode_name}")
                
    except KeyboardInterrupt:
        print("\n終了しました。")
    finally:
        # 必ず排他制御を解除して端末に制御を戻す
        try:
            dev.ungrab()
        except Exception:
            pass

if __name__ == '__main__':
    main()
